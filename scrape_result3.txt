	var INIT_MESSAGE = "";

	var USER_KEY = '';
	var DOMAIN_ID = '';
	var X = '';
	var Y = '';
	var ADDRESS = '';
	var GUBUN = '';
	var BOTNAME = '';
	var CHATBOT_PATH = "https://sgmbot.eseoul.go.kr:7614";

	var chat = {
		messageToSend: '',
		messageToShow: '',
		messageResponses: [],
		init: function(domain, user, scenario, x, y, address, gubun) {
			this.cacheDOM();
			this.bindEvents();
			Handlebars.registerHelper('if_eq', function(a, b, opts) {
				if (a === b) // Or === depending on your needs
					return opts.fn(this);
				else
					return opts.inverse(this);
			});

			USER_KEY = user;
			DOMAIN_ID = domain;
			X = x;
			Y = y;
			ADDRESS = address;
			GUBUN = gubun;
			var params = {};
			params.domain = domain;
			params.user = user;
			params.scenario = scenario;

			$.ajax({
				url: CHATBOT_PATH + "/konanbot/chat/open" ,
				dataType : "json",
				type : "POST",
				contentType : "application/json; charset=utf-8",
				data : JSON.stringify(params),
				processData : false
			})
					.done(function(ret){

						BOTNAME = ret.data.botname;
						answer(ret);
						
						// 최초 챗봇시 시간 위치 변경
						setTimeout(function() {
							$(".time").css("bottom", "-10px");
							$(".time").css("right", "-65px");
						}, 500);
						
					});
		},
		cacheDOM: function() {
			this.$chatHistory = $('#chat-history');
			this.$button = $('#send');
			this.$buttonInit= $('#init');
			this.$textarea = $('#textarea-b');
			this.$chatHistoryList = this.$chatHistory.find('ul');
		},
		bindEvents: function() {
			this.$button.on('click', this.addMessageEvent.bind(this));
			this.$textarea.on('keyup', this.addMessageEvent.bind(this));
		},
		question: function() {
			this.scrollToBottom();
			if ($.trim(this.messageToSend) !== '') {
				var template = Handlebars.compile($("#message-response-template").html());
				var time = this.getCurrentTime();
				if(time.split(":")[2]) {
					time = time.substring(0, time.length - 4);
				}
				var context = {
					messageOutput: this.messageToShow,
					time: time
				};

				this.$chatHistoryList.append(template(context));
				this.scrollToBottom();
			}

		},

		answer: function(msg) {
			msg = $.trim(msg);
			msg = msg.replace(/btn /g,'');
			var templateResponse = Handlebars.compile($("#message-template").html());
			var time = this.getCurrentTime();
			if(time.split(":")[2]) {
				time = time.substring(0, time.length - 4);
			}
			var contextResponse = {
				botname : BOTNAME,
				response: msg,
				time: time
			};
			setTimeout(function() {
				chat.$chatHistoryList.append(templateResponse(contextResponse));
				this.scrollToBottom();
			}.bind(this), 500);
		},

		sender: function(domain, user, x, y, address, gubun) {
			if (this.messageToSend === undefined || this.messageToSend === "") {
				return;
			}

			var params = {};
			params.domain = domain;
			params.user = user;
			params.x = x;
			params.y = y;
			params.address = address;
			params.gubun = gubun;
			params.text = this.messageToSend;

			$.ajax({
				url: CHATBOT_PATH + "/konanbot/chat/message",
				dataType : "json",
				type : "POST",
				contentType : "application/json; charset=utf-8",
				data : JSON.stringify(params),
				processData : false
			})
					.done(function(ret){
						console.log(ret);
						answer(ret);
					});

		},

		initMessage: function() {
			var msg = "";
			switch (KEY) {
				case KEY:
					msg = INIT_MESSAGE;
					break;
			}

			this.messageToSend = this.messageToShow = msg;
			this.question();
		},

		addMessage: function(msg) {

			//2020-05-19 메시지 중복성 출력 수정
			var chatConfirmMessage =$("#chatConfirmMessage").val();
			if(chatConfirmMessage==msg){
				return;
			}			
			$("#chatConfirmMessage").val(msg);			
			
			
			this.messageToSend = msg || $('#textarea-b').val();
			this.messageToShow = msg || $('#textarea-b').val();
			if ($.trim(this.messageToSend) === '') {
				return false;
			}
			this.$textarea.val('');
			this.question();
			this.sender(DOMAIN_ID, USER_KEY, X, Y, ADDRESS, GUBUN);
		},

		addMessageEvent: function(event) {
			// enter was pressed
			if (event.type === 'click' || event.keyCode === 13) {
				this.addMessage();
			}
		},
		scrollToBottom: function() {
			//$(".chat").getNiceScroll(0).doScrollTop($(".message").height(), 500);
			this.$chatHistory.animate({
				scrollTop: this.$chatHistory[0].scrollHeight
			}, 500);
			setTimeout(function() {
				this.$chatHistory.getNiceScroll().resize();
			}.bind(this));
		},
		getCurrentTime: function() {
			return new Date().toLocaleTimeString().
			replace(/([\d]+:[\d]{2})(:[\d]{2})(.*)/, "$1$3");
		},
		getRandomItem: function(arr) {
			return arr[Math.floor(Math.random() * arr.length)];
		}

	};


	function answer(data) {
		chat.answer(data.payload.contents[0].elements[0].text);
		/*
	    for (var i in data) {
	        chat.answer(data[i]);
	    }
	    */
	}

	const layer = document.querySelector('.chatbot-layer');
	const btnMin = document.querySelector('#btnChatMin, .btn-chat-min, .chatbot-min, .min'); 	// 최소화
	
	if(!layer || !btnMin) return;

	const span = btnMin.querySelector('span');
	const sr = btnMin.querySelector('aria_hidden');
	
	function syncMinButton() {
		const minimized = layer.classList.contains('min-on');
		const label =  minimized ? '챗봇 서비스 팝업 최대화' : '챗봇 서비스 팝업 최소화';
		
		btnMin.setAttribute('aria-label', label);
		
		// 내부 span/스크린리더 텍스트도 맞춰주기
		if(span) span.title = label;
		if(sr) 	 sr.textContect = label;
	}
	
	function toggleMin(e) {
		if(e){
			const k = e.key;
			if(k && k !== 'Enter' && k !== ' ') return;
			e.preventDefault();
		}
		syncMinButton();
	}
	
	btnMin.addEventListener('click', toggleMin);
	btnMin.addEventListener('keydown', (e)=>{
		if(e.key === 'Enter' || e.key === ' ') {e.preventDefault(); toggleMin(); }
	});

	// 외부 스크립트가 .min-on을 바꿔도 자동 반영
	new MutationObserver(syncMinButton).observe(layer, {attributes: true, attributeFilter: ['class']});
	
	// 초기 1회 반영
	syncMinButton();
	
}) ();

	function setCookie(name, value, expiredays){
	    var today = new Date();
	    today.setDate(today.getDate() + (expiredays || 3650));
	    document.cookie = name + '=' + escape(value) + '; expires=' + today.toGMTString();
	}
	
	function getCookie(name) {
	    var cookie = document.cookie;
	    if (document.cookie != "") {
	        var cookie_array = cookie.split("; ");
	        for ( var index in cookie_array) {
	            var cookie_name = cookie_array[index].split("=");
	            if (cookie_name[0] == name) {
	            	if(cookie_name[1] === "true") {
		                return true;
	            	} else {
		                return false;
	            	}
	            }
	        }
	    }
	    return false;
	}

    function getItems(menu){

		(function(){
			// 실행 조건: leftMenu.jsp의 contents 값이 'owner'일 때만
			var contentsVal = (typeof '{$contents}' === 'string') ? 'intendedOwner' : '';
			if (contentsVal !== 'owner') return;
			
			// 슬라이더 타켓 
			var sliders = Array.from(document.querySelectorAll('.linearRange'));
			if (!sliders.length) return;
			
			function isNativeRange(el){ return el.matches('input[type="range"]'); }
			function getVal(el) {
				// 값 읽기 (input=range or role=slider)
				return isNativeRange(el) ? el.value : (el.getAttribute('aria-valuenow') || '');
			}
			
			// 슬라이더별 설정 : id로 구분
			function getConfig(el){
				if(el.id === 'distanceRange') return { base: '보행거리', unit: 'KM' };
				if(el.id === 'timeRange') return { base: '보행시간', unit: '분' };
				
				// 혹시 다른 것이 섞여도 안전하게 기본값은 거리
				return { base: '보행거리', unit: 'KM' };
			}
					
			function setState(el, selected) {
				var cfg = getConfig(el);
				var val = getVal(el);
								
				// title
				el.title = selected ? '선택됨' : '선택안됨';
				
				// SR 보강
				el.setAttribute('aria-selected', selected ? 'true' : 'false');
				
				// 스크린리더용 라벨(현재값 포함)
				var valuePart = val ? (' ' + val +cfg.unit) : '';
				el.setAttribute('aria-label', cfg.base + valuePart + (selected ? '(선택됨)' : '(선택안됨)'));
			}
			
			function bind(el) {
				// 초기 상태 : 선택안됨
				setState(el, false);
			
				// 포커스 들어오면 선택됨
				el.addEventListener('focus', function(){ setState(el, true); });
				
				// 마우스로 드래그/터치 시작 시에도 선택됨
				['pointerdown', 'mousedown', 'tocuhstart'].forEach(function(evt){
					el.addEventListener(evt, function(){ setState(el, true);}, {passive:true});
				});
				
				// 값이 바뀌는 동안 유지
				['input', 'change'].forEach(function(evt){
					el.addEventListener(evt, function(){ setState(el, true);});
				});
				
				// 포커스 빠지면 산택 안됨
				el.addEventListener('blur', function(){ setState(el, false); });
				
				// 커스텀 슬라이더의 변경 감시
				var obs = new MutationObserver(function(muts){
					if (muts.some(m => m.attributeName === 'aria-valuenow')) {
						// 포커스 중이면 선택됨 유지, 아니면 선택안됨
						setState(el, document.attiveElement ===  el);
					}
				});
				obs.observe(el, { attributes:true, attributeFilter:['aria-valuenow']});
			}
			
			Array.from(document.querySelectorAll('.linearRange')).forEach(bind);
			
		})();

											매출을 선택하면 매출이 많은 행정동 또는 상권 순위를 확인할 수 있습니다.

							점포수

								개·폐업률 산식

								(당기 개/폐업신고점포수 / 전체점포수) * 100
							
						
					
					
					
					
					
						
							신생기업 생존율
							정의 : 1~5년 전 신생기업 중 '기준 연도'까지 생존 기업의 비율을 의미함
						

						
							
								생존은 기업체 신생 이후 t년까지 소멸되지 않고 지속적으로 존속한 경우를 의미함
								당해 연도 기업의 1~5년 생존율은 신생년도가 다르므로 n년 생존율이 n-1년 생존율 보다 작을 수 있음
								개·폐업이 거의 없는 업종은 생존율이 지나치게 높거나 낮을 수 있음
							
						

						
							
								(t)년도 생존율 산식

								1년 생존율 : (t-1)년 신생기업 중 (t)년까지 생존한 기업수 / (t-1)년 신생기업수 * 100
								5년 생존율 : (t-5)년 신생기업 중 (t)년까지 생존한 기업수 / (t-5)년 신생기업수 * 100
							
						
					
					
					
					
					
						
							연차별 생존율

							정의 : 과거(5년 전) 신생기업 중 1-5년 연차별 생존 기업의 비율을 의미함
						

						
							
								(t)년도 생존율 산식

								1년 생존율 : (t-5)년 신생기업 중 (t-5)+1년까지 생존한 기업수 / (t-5)년 신생기업수 * 100
								5년 생존율 : (t-5)년 신생기업 중 (t-5)+5년까지 생존한 기업수 / (t-5)년 신생기업수 * 100

								분석리포트에서는 최근 10년 기준으로 정보제공
